        "# =============================================================================\n",
        "# Sequence Generation (Sliding Window \u2014 Long-Horizon Forecasting)\n",
        "# =============================================================================\n",
        "# SEQ_LEN = 3 steps (6 hours lookback).\n",
        "# HORIZON = target step (48 = 4 days ahead, 72 = 6 days ahead).\n",
        "# For each position i: X = X[i : i+SEQ_LEN], y = y_raw[i + HORIZON].\n",
        "# Built independently per upazila to prevent cross-entity leakage.\n",
        "\n",
        "def create_sequences_long_horizon(X_scaled, y_raw, seq_len, horizon):\n",
        "    X_seq, y_seq = [], []\n",
        "    for i in range(len(X_scaled) - horizon):\n",
        "        X_seq.append(X_scaled[i : i + seq_len])\n",
        "        y_seq.append(y_raw[i + horizon])\n",
        "    return np.array(X_seq), np.array(y_seq)\n",
        "\n",
        "def build_sequences_upazila_wise(df_split, X_scaled_all, y_raw_all, seq_len, horizon):\n",
        "    X_all, y_all = [], []\n",
        "    for _, group in df_split.groupby(\"upazila\", sort=False):\n",
        "        idx = group.index\n",
        "        X_up = X_scaled_all[idx]\n",
        "        y_up = y_raw_all[idx]\n",
        "        X_s, y_s = create_sequences_long_horizon(X_up, y_up, seq_len, horizon)\n",
        "        X_all.extend(X_s)\n",
        "        y_all.extend(y_s)\n",
        "    return np.array(X_all), np.array(y_all)\n",
        "\n",
        "X_train_seq, y_train_seq = build_sequences_upazila_wise(\n",
        "    df_train, X_train_scaled, y_train_raw, SEQ_LEN, HORIZON\n",
        ")\n",
        "X_test_seq, y_test_seq = build_sequences_upazila_wise(\n",
        "    df_test, X_test_scaled, y_test_raw, SEQ_LEN, HORIZON\n",
        ")\n",
        "\n",
        "# Chronological validation split (last 15% of training sequences)\n",
        "VAL_RATIO = 0.15\n",
        "val_split_idx = int(len(X_train_seq) * (1 - VAL_RATIO))\n",
        "X_val_seq, y_val_seq = X_train_seq[val_split_idx:], y_train_seq[val_split_idx:]\n",
        "X_train_seq, y_train_seq = X_train_seq[:val_split_idx], y_train_seq[:val_split_idx]\n",
